`packages/kernel/src/testing/memory-storage.ts`:
```ts
import { randomUUID } from "node:crypto";
import { messageText, type ModelMessage } from "../messages.js";
import type { SearchHit, SessionMeta, StoragePort } from "../ports.js";

export class InMemoryStorage implements StoragePort {
  private readonly sessions = new Map<string, SessionMeta>();
  private readonly messages = new Map<string, ModelMessage[]>();

  async createSession(title: string): Promise<SessionMeta> {
    const now = new Date().toISOString();
    const meta: SessionMeta = { id: randomUUID(), title, createdAt: now, updatedAt: now };
    this.sessions.set(meta.id, meta);
    this.messages.set(meta.id, []);
    return meta;
  }

  async getSession(id: string): Promise<SessionMeta | null> {
    return this.sessions.get(id) ?? null;
  }

  async listSessions(limit = 50): Promise<SessionMeta[]> {
    return [...this.sessions.values()]
      .sort((a, b) => b.updatedAt.localeCompare(a.updatedAt))
      .slice(0, limit);
  }

  async appendMessages(sessionId: string, messages: ModelMessage[]): Promise<void> {
    const existing = this.messages.get(sessionId);
    if (!existing) throw new Error(`unknown session: ${sessionId}`);
    existing.push(...structuredClone(messages));
    const meta = this.sessions.get(sessionId);
    if (meta) meta.updatedAt = new Date().toISOString();
  }

  async loadMessages(sessionId: string): Promise<ModelMessage[]> {
    return structuredClone(this.messages.get(sessionId) ?? []);
  }

  async search(query: string, limit = 20): Promise<SearchHit[]> {
    const needle = query.toLowerCase();
    const hits: SearchHit[] = [];
    for (const [sessionId, messages] of this.messages) {
      for (const message of messages) {
        const text = messageText(message);
        if (text.toLowerCase().includes(needle)) hits.push({ sessionId, snippet: text });
        if (hits.length >= limit) return hits;
      }
    }
    return hits;
  }
}
```

注意 `listSessions` 排序键是 ISO 字符串——同毫秒内连续创建会并列，测试里用 appendMessages 制造顺序差异而非依赖创建时间；如测试仍偶发并列失败，在 InMemoryStorage.updatedAt 比较相等时按插入序回退（Map 迭代序即插入序）。

`packages/storage/src/migrations.ts`:
```ts
import type Database from "better-sqlite3";

export type Migration = { version: number; sql: string };

export function applyMigrations(
  db: Database.Database,
  migrations: Migration[],
): number {
  db.exec(
    "CREATE TABLE IF NOT EXISTS schema_migrations (version INTEGER PRIMARY KEY, applied_at TEXT NOT NULL)",
  );
  const appliedRows = db.prepare("SELECT version FROM schema_migrations").all();
  const applied = new Set(appliedRows.map((r) => (r as { version: number }).version));
  let count = 0;
  for (const migration of [...migrations].sort((a, b) => a.version - b.version)) {
    if (applied.has(migration.version)) continue;
    db.transaction(() => {
      db.exec(migration.sql);
      db.prepare("INSERT INTO schema_migrations (version, applied_at) VALUES (?, ?)").run(
        migration.version,
        new Date().toISOString(),
      );
    })();
    count++;
  }
  return count;
}
```

import { randomUUID } from "node:crypto";
import type { ModelMessage, SearchHit, SessionMeta, StoragePort } from "@hermes-ts/kernel";
import Database from "better-sqlite3";
import { applyMigrations } from "./migrations.js";
import { MIGRATIONS } from "./schema.js";

type SessionRow = { id: string; title: string; created_at: string; updated_at: string };

function toMeta(row: SessionRow): SessionMeta {
  return { id: row.id, title: row.title, createdAt: row.created_at, updatedAt: row.updated_at };
}

export class SqliteStorage implements StoragePort {
  private readonly db: Database.Database;

  constructor(path: string) {
    this.db = new Database(path);
    this.db.pragma("journal_mode = WAL");
    applyMigrations(this.db, MIGRATIONS);
  }

  close(): void {
    this.db.close();
  }

  async createSession(title: string): Promise<SessionMeta> {
    const now = new Date().toISOString();
    const meta: SessionMeta = { id: randomUUID(), title, createdAt: now, updatedAt: now };
    this.db
      .prepare("INSERT INTO sessions (id, title, created_at, updated_at) VALUES (?, ?, ?, ?)")
      .run(meta.id, meta.title, meta.createdAt, meta.updatedAt);
    return meta;
  }

  async getSession(id: string): Promise<SessionMeta | null> {
    const row = this.db.prepare("SELECT * FROM sessions WHERE id = ?").get(id) as
      | SessionRow
      | undefined;
    return row ? toMeta(row) : null;
  }

  async listSessions(limit = 50): Promise<SessionMeta[]> {
    const rows = this.db
      .prepare("SELECT * FROM sessions ORDER BY updated_at DESC, rowid DESC LIMIT ?")
      .all(limit) as SessionRow[];
    return rows.map(toMeta);
  }

  async appendMessages(_sessionId: string, _messages: ModelMessage[]): Promise<void> {

  }

  async loadMessages(_sessionId: string): Promise<ModelMessage[]> {
    throw new Error("not implemented"); // Task 4
  }

  async search(_query: string, _limit = 20): Promise<SearchHit[]> {

`packages/storage/src/sqlite-storage.ts` 中替换 `appendMessages`/`loadMessages`:
```ts
  async appendMessages(sessionId: string, messages: ModelMessage[]): Promise<void> {
    const session = await this.getSession(sessionId);
    if (!session) throw new Error(`unknown session: ${sessionId}`);
    const now = new Date().toISOString();
    const insert = this.db.prepare(
      "INSERT INTO messages (session_id, idx, content_json, text_content, created_at) VALUES (?, ?, ?, ?, ?)",
    );
    const nextIdxRow = this.db
      .prepare("SELECT COALESCE(MAX(idx), -1) + 1 AS next FROM messages WHERE session_id = ?")
      .get(sessionId) as { next: number };
    this.db.transaction(() => {
      let idx = nextIdxRow.next;
      for (const message of messages) {
        insert.run(sessionId, idx, JSON.stringify(message), messageTextOf(message), now);
        idx++;
      }
      this.db.prepare("UPDATE sessions SET updated_at = ? WHERE id = ?").run(now, sessionId);
    })();
  }

  async loadMessages(sessionId: string): Promise<ModelMessage[]> {
    const rows = this.db
      .prepare("SELECT content_json FROM messages WHERE session_id = ? ORDER BY idx ASC")
      .all(sessionId) as { content_json: string }[];
    return rows.map((row) => JSON.parse(row.content_json) as ModelMessage);
  }
```

```ts
import { readFile } from "node:fs/promises";
import { homedir } from "node:os";
import { join } from "node:path";
import { parse } from "yaml";
import { configSchema, DEFAULT_MODELS, type HermesConfig } from "./schema.js";

export function defaultConfigDir(): string {
  return join(homedir(), ".hermes-ts");
}

async function readYamlObject(path: string): Promise<Record<string, unknown>> {
  let text: string;
  try {
    text = await readFile(path, "utf8");
  } catch (error) {
    if ((error as NodeJS.ErrnoException).code === "ENOENT") return {};
    throw error;
  }
  const value = parse(text) ?? {};
  if (typeof value !== "object" || Array.isArray(value)) {
    throw new Error(`配置文件必须是 YAML mapping: ${path}`);
  }
  return value as Record<string, unknown>;
}

function envOverrides(env: Record<string, string | undefined>): Record<string, unknown> {
  return {
    ...(env.HERMES_PROVIDER ? { provider: env.HERMES_PROVIDER } : {}),
    ...(env.HERMES_MODEL ? { model: env.HERMES_MODEL } : {}),
    ...(env.HERMES_BASE_URL ? { baseUrl: env.HERMES_BASE_URL } : {}),
    ...(env.HERMES_MAX_ITERATIONS ? { maxIterations: Number(env.HERMES_MAX_ITERATIONS) } : {}),
    ...(env.HERMES_STORAGE_PATH ? { storagePath: env.HERMES_STORAGE_PATH } : {}),
  };
}

export async function loadConfig(
  options: { configPath?: string; env?: Record<string, string | undefined> } = {},
): Promise<HermesConfig> {
  const env = options.env ?? process.env;
  const configPath = options.configPath ?? join(defaultConfigDir(), "config.yaml");
  const merged = { ...(await readYamlObject(configPath)), ...envOverrides(env) };
  const parsed = configSchema.safeParse(merged);
  if (!parsed.success) {
    const issues = parsed.error.issues
      .map((issue) => `  ${issue.path.join(".") || "(root)"}: ${issue.message}`)
      .join("\n");
    throw new Error(`配置无效 (${configPath}):\n${issues}`);
  }
  return {
    ...parsed.data,
    model: parsed.data.model ?? DEFAULT_MODELS[parsed.data.provider],
    storagePath: parsed.data.storagePath ?? join(defaultConfigDir(), "hermes.db"),
  };
}

import type { ModelMessage, ToolDefinition } from "@hermes-ts/kernel";
import type { MessageParam, Tool } from "@anthropic-ai/sdk/resources/messages";

export function toAnthropicSystem(messages: ModelMessage[]): string | undefined {
  const system = messages.find((m) => m.role === "system");
  return system?.role === "system" ? system.content : undefined;
}

export function toAnthropicMessages(messages: ModelMessage[]): MessageParam[] {
  const result: MessageParam[] = [];
  for (const message of messages) {
    if (message.role === "system") continue;
    if (message.role === "user") {
      result.push({
        role: "user",
        content: message.content.map((part) => ({ type: "text" as const, text: part.text })),
      });
    } else if (message.role === "assistant") {
      const blocks = message.content.map((part) =>
        part.type === "text"
          ? { type: "text" as const, text: part.text }
          : {
              type: "tool_use" as const,
              id: part.toolCallId,
              name: part.toolName,
              input: part.args,
            },
      );
      if (blocks.length === 0) continue; // Anthropic 拒绝空内容块数组（M1 终审同类教训）
      result.push({ role: "assistant", content: blocks });
    } else {
      result.push({
        role: "user",
        content: message.content.map((part) => ({
          type: "tool_result" as const,
          tool_use_id: part.toolCallId,
          content: part.output,
          ...(part.isError ? { is_error: true } : {}),
        })),
      });
    }
  }
  return result;
}

export function toAnthropicTools(tools: ToolDefinition[]): Tool[] {
  return tools.map((tool) => ({
    name: tool.name,
    description: tool.description,
    // ToolDefinition.parameters 是 zod 产出的 JSON Schema（object 型）；
    // SDK 的 InputSchema 要求判别字段 type:"object"，运行时形状一致，窄域断言。
    input_schema: tool.parameters as Tool["input_schema"],
  }));
}
```

import type { ProviderEvent } from "@hermes-ts/kernel";

export type AnthropicStreamEvent = {
  type: string;
  index?: number;
  content_block?: { type: string; id?: string; name?: string };
  delta?: {
    type?: string;
    text?: string;
    partial_json?: string;
    stop_reason?: string | null;
  };
};

type PendingCall = { id: string; name: string; jsonText: string };

export async function* eventsFromAnthropicStream(
  events: AsyncIterable<AnthropicStreamEvent>,
): AsyncIterable<ProviderEvent> {
  const pending = new Map<number, PendingCall>();
  let stopReason: "stop" | "tool-calls" = "stop";

  for await (const event of events) {
    if (
      event.type === "content_block_start" &&
      event.content_block?.type === "tool_use" &&
      event.index !== undefined
    ) {
      pending.set(event.index, {
        id: event.content_block.id ?? "",
        name: event.content_block.name ?? "",
        jsonText: "",
      });
    } else if (event.type === "content_block_delta") {
      if (event.delta?.type === "text_delta" && event.delta.text) {
        yield { type: "text-delta", text: event.delta.text };
      } else if (event.delta?.type === "input_json_delta" && event.index !== undefined) {
        const call = pending.get(event.index);
        if (call && event.delta.partial_json) call.jsonText += event.delta.partial_json;
      }
    } else if (event.type === "message_delta" && event.delta?.stop_reason === "tool_use") {
      stopReason = "tool-calls";
    }
  }

  for (const [, call] of [...pending.entries()].sort(([a], [b]) => a - b)) {
    // 与 OpenAI 拼装层一致：畸形 JSON 在此抛出，上游 Agent 捕获为 error 事件（有意为之）
    yield {
      type: "tool-call",
      toolCallId: call.id,
      toolName: call.name,
      args: call.jsonText ? JSON.parse(call.jsonText) : {},
    };
  }
  yield { type: "finish", stopReason };
}

import type { ProviderEvent, ProviderPort, ProviderRequest } from "@hermes-ts/kernel";
import { toAnthropicMessages, toAnthropicSystem, toAnthropicTools } from "./mapping.js";
import { eventsFromAnthropicStream } from "./stream.js";

const DEFAULT_MAX_TOKENS = 8192;

export class AnthropicProvider implements ProviderPort {
  private readonly client: Anthropic;

  constructor(options: { apiKey: string; baseUrl?: string }) {
    // spec §8 的 provider 重试/退避由 SDK 内建机制承担（指数退避，408/429/5xx/网络错误）。
    // 流已开始后中断不重试：重放可能重复工具调用等副作用，交由 Agent 的 error 事件浮出。
    this.client = new Anthropic({
      apiKey: options.apiKey,
      baseURL: options.baseUrl,
      maxRetries: 4,
    });
  }

  async *stream(request: ProviderRequest): AsyncIterable<ProviderEvent> {
    const system = toAnthropicSystem(request.messages);
    const stream = await this.client.messages.create(
      {
        model: request.model,
        max_tokens: DEFAULT_MAX_TOKENS,
        ...(system ? { system } : {}),
        messages: toAnthropicMessages(request.messages),
        ...(request.tools.length > 0 ? { tools: toAnthropicTools(request.tools) } : {}),
        stream: true,
      },
      { signal: request.signal },
    );
    yield* eventsFromAnthropicStream(stream);
  }
}

import type { ModelMessage, StoragePort } from "@hermes-ts/kernel";

export type SessionHandle = { id: string; history: ModelMessage[] };

export async function openSession(
  storage: StoragePort,
  flags: { newSession?: boolean; sessionId?: string },
  title: string,
): Promise<SessionHandle> {
  if (flags.sessionId) {
    const meta = await storage.getSession(flags.sessionId);
    if (!meta) throw new Error(`会话不存在: ${flags.sessionId}`);
    return { id: meta.id, history: await storage.loadMessages(meta.id) };
  }
  if (!flags.newSession) {
    const [latest] = await storage.listSessions(1);
    if (latest) return { id: latest.id, history: await storage.loadMessages(latest.id) };
  }
  const meta = await storage.createSession(title);
  return { id: meta.id, history: [] };
}

export async function persistTurn(
  storage: StoragePort,
  handle: SessionHandle,
  messages: ModelMessage[],
): Promise<void> {
  const fresh = messages.slice(handle.history.length);
  if (fresh.length > 0) await storage.appendMessages(handle.id, fresh);
  handle.history = messages;
}
