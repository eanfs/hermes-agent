**Architecture:** 六边形架构：`kernel` 定义端口（ProviderPort/ToolPort）与事件流，Agent 循环是一个异步生成器；`tools`、`providers`、`cli` 都是端口的实现或事件的消费者。测试以 FakeProvider 驱动，零网络。

**Tech Stack:** Node ≥22、pnpm workspaces、TypeScript strict (ESM)、Vitest、Biome、zod v4（`z.toJSONSchema`）、openai SDK、tsx。

```ts
export type TextPart = { type: "text"; text: string };
export type ToolCallPart = {
  type: "tool-call";
  toolCallId: string;
  toolName: string;
  args: unknown;
};
export type ToolResultPart = {
  type: "tool-result";
  toolCallId: string;
  toolName: string;
  output: string;
  isError: boolean;
};

export type SystemMessage = { role: "system"; content: string };
export type UserMessage = { role: "user"; content: TextPart[] };
export type AssistantMessage = {
  role: "assistant";
  content: (TextPart | ToolCallPart)[];
};
export type ToolMessage = { role: "tool"; content: ToolResultPart[] };
export type ModelMessage =
  | SystemMessage
  | UserMessage
  | AssistantMessage
  | ToolMessage;

export function userMessage(text: string): UserMessage {
  return { role: "user", content: [{ type: "text", text }] };
}

export function messageText(message: ModelMessage): string {
  if (message.role === "system") return message.content;
  return message.content
    .filter((part): part is TextPart => part.type === "text")
    .map((part) => part.text)
    .join("");
}
```

```ts
import type { ProviderEvent, ProviderPort, ProviderRequest } from "../ports.js";

export class FakeProvider implements ProviderPort {
  readonly requests: ProviderRequest[] = [];
  private readonly turns: ProviderEvent[][];

  constructor(turns: ProviderEvent[][]) {
    this.turns = [...turns];
  }

  async *stream(request: ProviderRequest): AsyncIterable<ProviderEvent> {
    if (request.signal?.aborted) throw new Error("aborted");
    this.requests.push(request);
    const turn = this.turns.shift();
    if (!turn) throw new Error("FakeProvider: no scripted turn left");
    yield* turn;
  }
}
```

`packages/tools/src/registry.ts`:
```ts
import type { ToolExecution, ToolPort } from "@hermes-ts/kernel";
import { z } from "zod";

export type ToolContext = { cwd: string; signal: AbortSignal };

export type Tool<Schema extends z.ZodType = z.ZodType> = {
  name: string;
  description: string;
  parameters: Schema;
  execute(args: z.infer<Schema>, ctx: ToolContext): Promise<string>;
};

export function defineTool<S extends z.ZodType>(tool: Tool<S>): Tool<S> {
  return tool;
}

export function createToolRegistry(
  tools: Tool[],
  options: { cwd: string },
): ToolPort {
  const byName = new Map(tools.map((tool) => [tool.name, tool]));
  if (byName.size !== tools.length) throw new Error("duplicate tool name");

  return {
    definitions: () =>
      tools.map((tool) => ({
        name: tool.name,
        description: tool.description,
        parameters: z.toJSONSchema(tool.parameters) as Record<string, unknown>,
      })),

    async execute(name, args, signal): Promise<ToolExecution> {
      const tool = byName.get(name);
      if (!tool) return { output: `Unknown tool: ${name}`, isError: true };
      const parsed = tool.parameters.safeParse(args);
      if (!parsed.success) {
        return { output: `Invalid arguments: ${parsed.error.message}`, isError: true };
      }
      try {
        const output = await tool.execute(parsed.data, { cwd: options.cwd, signal });
        return { output, isError: false };
      } catch (error) {
        const message = error instanceof Error ? error.message : String(error);
        return { output: message, isError: true };
      }
    },
  };
}
```

```ts
import { promisify } from "node:util";
import { z } from "zod";
import { defineTool } from "../registry.js";

const execFileAsync = promisify(execFile);
const TIMEOUT_MS = 60_000;

export const shellTool = defineTool({
  name: "shell",
  description: "Run a shell command in the working directory. Returns stdout+stderr.",
  parameters: z.object({ command: z.string().describe("Shell command to run") }),
  execute: async (args, ctx) => {
    try {
      const { stdout, stderr } = await execFileAsync("/bin/sh", ["-c", args.command], {
        cwd: ctx.cwd,
        signal: ctx.signal,
        timeout: TIMEOUT_MS,
        maxBuffer: 1024 * 1024,
      });
      return stdout + stderr;
    } catch (error) {
      const err = error as NodeJS.ErrnoException & {
        code?: number | string;
        stdout?: string;
        stderr?: string;
        killed?: boolean;
      };
      const output = `${err.stdout ?? ""}${err.stderr ?? ""}`;
      if (err.killed) throw new Error(`command timed out after ${TIMEOUT_MS}ms\n${output}`);
      throw new Error(`command failed with exit code ${err.code}\n${output}`);
    }
  },
});
```

```ts
import type { AgentEvent } from "./events.js";
import {
  type AssistantMessage,
  type ModelMessage,
  type TextPart,
  type ToolCallPart,
  type ToolMessage,
  type ToolResultPart,
  userMessage,
} from "./messages.js";
import type { ProviderPort, ToolDefinition, ToolPort } from "./ports.js";

export type AgentConfig = {
  model: string;
  systemPrompt?: string;
  maxIterations?: number;
};

export type AgentPorts = { provider: ProviderPort; tools: ToolPort };

export class Agent {
  constructor(
    private readonly config: AgentConfig,
    private readonly ports: AgentPorts,
  ) {}

  async *runTurn(
    input: string,
    history: ModelMessage[] = [],
    signal: AbortSignal = new AbortController().signal,
  ): AsyncGenerator<AgentEvent> {
    const maxIterations = this.config.maxIterations ?? 10;
    const messages: ModelMessage[] = [...history];
    if (this.config.systemPrompt && !messages.some((m) => m.role === "system")) {
      messages.unshift({ role: "system", content: this.config.systemPrompt });
    }
    messages.push(userMessage(input));
    yield { type: "turn-start" };

    try {
      for (let iteration = 0; iteration < maxIterations; iteration++) {
        if (signal.aborted) throw new Error("aborted");
        const { assistant, toolCalls } = yield* this.streamOnce(
          messages,
          this.ports.tools.definitions(),
          signal,
        );
        messages.push(assistant);
        if (toolCalls.length === 0) {
          yield { type: "turn-complete", messages };
          return;
        }
        messages.push(yield* this.executeTools(toolCalls, signal));
      }

      yield { type: "budget-exhausted", iterations: maxIterations };
      const { assistant } = yield* this.streamOnce(messages, [], signal);
      messages.push(assistant);
      yield { type: "turn-complete", messages };
    } catch (error) {
      const message = error instanceof Error ? error.message : String(error);
      yield { type: "error", message };
    }
  }

  private async *streamOnce(
    messages: ModelMessage[],
    tools: ToolDefinition[],
    signal: AbortSignal,
  ): AsyncGenerator<AgentEvent, { assistant: AssistantMessage; toolCalls: ToolCallPart[] }> {
    let text = "";
    const toolCalls: ToolCallPart[] = [];
    const request = { model: this.config.model, messages, tools, signal };
    for await (const event of this.ports.provider.stream(request)) {
      if (event.type === "text-delta") {
        text += event.text;
        yield { type: "text-delta", text: event.text };
      } else if (event.type === "tool-call") {
        toolCalls.push({
          type: "tool-call",
          toolCallId: event.toolCallId,
          toolName: event.toolName,
          args: event.args,
        });
      }
    }
    const textParts: TextPart[] = text ? [{ type: "text", text }] : [];
    return {
      assistant: { role: "assistant", content: [...textParts, ...toolCalls] },
      toolCalls,
    };
  }

  private async *executeTools(
    toolCalls: ToolCallPart[],
    signal: AbortSignal,
  ): AsyncGenerator<AgentEvent, ToolMessage> {
    for (const call of toolCalls) {
      yield {
        type: "tool-call-start",
        toolCallId: call.toolCallId,
        toolName: call.toolName,
        args: call.args,
      };
    }
    const results = await Promise.all(
      toolCalls.map((call) => this.ports.tools.execute(call.toolName, call.args, signal)),
    );
    const parts: ToolResultPart[] = [];
    for (const [index, call] of toolCalls.entries()) {
      const result = results[index];
      if (!result) continue;
      yield {
        type: "tool-call-end",
        toolCallId: call.toolCallId,
        toolName: call.toolName,
        output: result.output,
        isError: result.isError,
      };
      parts.push({
        type: "tool-result",
        toolCallId: call.toolCallId,
        toolName: call.toolName,
        output: result.output,
        isError: result.isError,
      });
    }
    return { role: "tool", content: parts };
  }
}
```

```ts
import type { ProviderEvent } from "@hermes-ts/kernel";

export type OpenAiChunk = {
  choices: Array<{
    delta: {
      content?: string | null;
      tool_calls?: Array<{
        index: number;
        id?: string;
        function?: { name?: string; arguments?: string };
      }>;
    };
    finish_reason?: string | null;
  }>;
};

type PendingCall = { id: string; name: string; argsText: string };

export async function* streamFromChunks(
  chunks: AsyncIterable<OpenAiChunk>,
): AsyncIterable<ProviderEvent> {
  const pending = new Map<number, PendingCall>();
  let stopReason: "stop" | "tool-calls" = "stop";

  for await (const chunk of chunks) {
    const choice = chunk.choices[0];
    if (!choice) continue;
    if (choice.delta.content) {
      yield { type: "text-delta", text: choice.delta.content };
    }
    for (const call of choice.delta.tool_calls ?? []) {
      const entry = pending.get(call.index) ?? { id: "", name: "", argsText: "" };
      if (call.id) entry.id = call.id;
      if (call.function?.name) entry.name = call.function.name;
      if (call.function?.arguments) entry.argsText += call.function.arguments;
      pending.set(call.index, entry);
    }
    if (choice.finish_reason === "tool_calls") stopReason = "tool-calls";
  }

  for (const [, call] of [...pending.entries()].sort(([a], [b]) => a - b)) {
    yield {
      type: "tool-call",
      toolCallId: call.id,
      toolName: call.name,
      args: call.argsText ? JSON.parse(call.argsText) : {},
    };
  }
  yield { type: "finish", stopReason };
}
```

`packages/providers/src/openai/mapping.ts`:
```ts
import type { ModelMessage, ToolDefinition } from "@hermes-ts/kernel";
import { messageText } from "@hermes-ts/kernel";
import type {
  ChatCompletionMessageParam,
  ChatCompletionTool,
} from "openai/resources/chat/completions";

export function toOpenAiMessages(messages: ModelMessage[]): ChatCompletionMessageParam[] {
  const result: ChatCompletionMessageParam[] = [];
  for (const message of messages) {
    if (message.role === "system") {
      result.push({ role: "system", content: message.content });
    } else if (message.role === "user") {
      result.push({ role: "user", content: messageText(message) });
    } else if (message.role === "assistant") {
      const toolCalls = message.content
        .filter((part) => part.type === "tool-call")
        .map((part) => ({
          id: part.toolCallId,
          type: "function" as const,
          function: { name: part.toolName, arguments: JSON.stringify(part.args) },
        }));
      const text = messageText(message);
      result.push({
        role: "assistant",
        ...(text ? { content: text } : { content: null }),
        ...(toolCalls.length > 0 ? { tool_calls: toolCalls } : {}),
      });
    } else {
      for (const part of message.content) {
        result.push({ role: "tool", tool_call_id: part.toolCallId, content: part.output });
      }
    }
  }
  return result;
}

export function toOpenAiTools(tools: ToolDefinition[]): ChatCompletionTool[] {
  return tools.map((tool) => ({
    type: "function",
    function: {
      name: tool.name,
      description: tool.description,
      parameters: tool.parameters,
    },
  }));
}
```

注意：`omits empty assistant content` 测试期望 `{ role: "assistant", content: "ok" }` 不带 `content: null` 分支——实现里 `text` 非空时用 `content: text`，为空时用 `content: null`（OpenAI 要求 assistant 消息在有 tool_calls 时 content 可为 null）。若测试因 `content: null` 键存在而失败，按测试期望修正实现（测试是真相源）。

`packages/providers/src/openai/provider.ts`:
```ts
import type { ProviderEvent, ProviderPort, ProviderRequest } from "@hermes-ts/kernel";
import OpenAI from "openai";
import { streamFromChunks } from "./chunks.js";
import { toOpenAiMessages, toOpenAiTools } from "./mapping.js";

export class OpenAiProvider implements ProviderPort {
  private readonly client: OpenAI;

  constructor(options: { apiKey: string; baseUrl?: string }) {
    this.client = new OpenAI({ apiKey: options.apiKey, baseURL: options.baseUrl });
  }

  async *stream(request: ProviderRequest): AsyncIterable<ProviderEvent> {
    const stream = await this.client.chat.completions.create(
      {
        model: request.model,
        messages: toOpenAiMessages(request.messages),
        ...(request.tools.length > 0 ? { tools: toOpenAiTools(request.tools) } : {}),
        stream: true,
      },
      { signal: request.signal },
    );
    yield* streamFromChunks(stream);
  }
}
```

`apps/cli/src/render.ts`:
```ts
import type { AgentEvent } from "@hermes-ts/kernel";

const MAX_OUTPUT_PREVIEW = 200;

export function renderEvent(event: AgentEvent): string {
  switch (event.type) {
    case "text-delta":
      return event.text;
    case "tool-call-start":
      return `\n⚙ ${event.toolName} ${JSON.stringify(event.args)}\n`;
    case "tool-call-end": {
      const preview = event.output.slice(0, MAX_OUTPUT_PREVIEW).trimEnd();
      return `${event.isError ? "✗" : "✓"} ${event.toolName}: ${preview}\n`;
    }
    case "budget-exhausted":
      return `\n[已达迭代上限 ${event.iterations}，收尾中]\n`;
    case "error":
      return `\n[error] ${event.message}\n`;
    default:
      return "";
  }
}
```

```ts
import readline from "node:readline/promises";
import { Agent, type ModelMessage } from "@hermes-ts/kernel";
import { OpenAiProvider } from "@hermes-ts/providers";
import { createToolRegistry, readFileTool, shellTool, writeFileTool } from "@hermes-ts/tools";
import { renderEvent } from "./render.js";

const apiKey = process.env.OPENAI_API_KEY;
if (!apiKey) {
  console.error("OPENAI_API_KEY 未设置。请 export OPENAI_API_KEY=... 后重试。");
  process.exit(1);
}

const provider = new OpenAiProvider({ apiKey, baseUrl: process.env.OPENAI_BASE_URL });
const tools = createToolRegistry([shellTool, readFileTool, writeFileTool], {
  cwd: process.cwd(),
});
const agent = new Agent(
  {
    model: process.env.HERMES_MODEL ?? "gpt-4o-mini",
    systemPrompt:
      "You are Hermes, a helpful agent running in the user's terminal. " +
      "Use the shell/read_file/write_file tools when they help. Be concise.",
  },
  { provider, tools },
);

const rl = readline.createInterface({ input: process.stdin, output: process.stdout });
let history: ModelMessage[] = [];

console.log("hermes-ts M1 — 输入消息开始对话，/exit 退出。");
for (;;) {
  const input = (await rl.question("\n> ")).trim();
  if (input === "/exit") break;
  if (!input) continue;
  for await (const event of agent.runTurn(input, history)) {
    process.stdout.write(renderEvent(event));
    if (event.type === "turn-complete") history = event.messages;
  }
  process.stdout.write("\n");
}
rl.close();
```
