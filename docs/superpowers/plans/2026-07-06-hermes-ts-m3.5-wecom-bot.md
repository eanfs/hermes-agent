  - 出站回复：`{ cmd: "aibot_respond_msg", headers: { req_id: <入站 req_id> }, body: { msgtype: "text", text: { content } } }`（无 chatid，靠 req_id 路由）
  - **群聊约束**：AI Bot 群里不能主动 `aibot_send_msg`，必须 `aibot_respond_msg` 绑定入站 req_id
  - 心跳：每 30s 发 `{ cmd: "ping", headers: { req_id: "ping-<uuid>" }, body: {} }`；服务器 ping 忽略
  - 重连退避 `[2,5,10,30,60]`s，重连时自动重发 subscribe；dedup by msgid TTL 300s，重连不清
  - MAX_MESSAGE_LENGTH 4000（markdown/text content 上限；本计划用 text，超长分段）
  - req_id 格式 `"<prefix>-<uuid4hex>"`
- v1 用 **text** 消息类型（非 markdown，与 M3 回调版一致，纯文本稳妥）；v1 非目标见末尾

`packages/channel-wecom-bot/src/protocol.ts`:
```ts
const MAX_LEN = 4000;
const CALLBACK_CMDS = new Set(["aibot_msg_callback", "aibot_callback"]);

export type Envelope = {
  cmd: string;
  headers: { req_id: string };
  body: Record<string, unknown>;
};

export type BotInbound = {
  msgId: string;
  fromUser: string;
  chatId: string;
  isGroup: boolean;
  text: string;
  reqId: string;
} | null;

export function newReqId(prefix: string, uuid: () => string): string {
  return `${prefix}-${uuid()}`;
}

export function subscribeFrame(botId: string, secret: string, deviceId: string, reqId: string): Envelope {
  return { cmd: "aibot_subscribe", headers: { req_id: reqId }, body: { bot_id: botId, secret, device_id: deviceId } };
}

export function pingFrame(reqId: string): Envelope {
  return { cmd: "ping", headers: { req_id: reqId }, body: {} };
}

export function sendMsgFrame(chatId: string, text: string, reqId: string): Envelope {
  return {
    cmd: "aibot_send_msg",
    headers: { req_id: reqId },
    body: { chatid: chatId, msgtype: "text", text: { content: text.slice(0, MAX_LEN) } },
  };
}

export function respondMsgFrame(inboundReqId: string, text: string): Envelope {
  return {
    cmd: "aibot_respond_msg",
    headers: { req_id: inboundReqId },
    body: { msgtype: "text", text: { content: text.slice(0, MAX_LEN) } },
  };
}

function str(v: unknown): string {
  return typeof v === "string" ? v : "";
}

export function parseInbound(env: Envelope): BotInbound {
  if (!CALLBACK_CMDS.has(env.cmd)) return null;
  const body = env.body as {
    msgid?: string;
    from?: { userid?: string };
    chatid?: string;
    chattype?: string;
    msgtype?: string;
    text?: { content?: string };
  };
  if (body.msgtype !== "text") return null;
  const fromUser = str(body.from?.userid);
  const isGroup = body.chattype === "group";
  let text = str(body.text?.content);
  if (isGroup) text = text.replace(/^@\S+\s*/, "");
  return {
    msgId: str(body.msgid) || env.headers.req_id,
    fromUser,
    chatId: str(body.chatid) || fromUser,
    isGroup,
    text,
    reqId: env.headers.req_id,
  };
}
```

`packages/channel-wecom-bot/src/reply-tracker.ts`:
```ts
import { type Envelope, respondMsgFrame, sendMsgFrame } from "./protocol.js";

export class ReplyTracker {
  private readonly lastReqIdByChat = new Map<string, string>();
  private readonly max: number;

  constructor(max = 1000) {
    this.max = max;
  }

  remember(inbound: { msgId: string; chatId: string; reqId: string }): void {
    // Map 迭代序即插入序；重设 key 时先删再插以刷新其新鲜度
    this.lastReqIdByChat.delete(inbound.chatId);
    this.lastReqIdByChat.set(inbound.chatId, inbound.reqId);
    while (this.lastReqIdByChat.size > this.max) {
      const oldest = this.lastReqIdByChat.keys().next().value;
      if (oldest === undefined) break;
      this.lastReqIdByChat.delete(oldest);
    }
  }

  frameFor(chatId: string, text: string, newReqId: () => string): Envelope {
    const reqId = this.lastReqIdByChat.get(chatId);
    if (reqId) return respondMsgFrame(reqId, text);
    return sendMsgFrame(chatId, text, newReqId());
  }
}
```

**Interfaces:**
- Produces:
  - `interface SocketLike { send(data: string): void; close(): void; on(event: "message" | "open" | "close" | "error", cb: (arg?: unknown) => void): void }`
  - `type SocketFactory = (url: string) => SocketLike`
  - `class BotConnection { constructor(opts: { url: string; botId: string; secret: string; deviceId: string; socketFactory: SocketFactory; uuid: () => string; onInbound: (env: Envelope) => void; onError?: (e: Error) => void }); connect(): Promise<void>; sendRequest(frame: Envelope, timeoutMs?: number): Promise<Envelope>; sendFrame(frame: Envelope): void; close(): void }`
  - connect：new socket → open 后发 subscribe → 等 headers.req_id 匹配且 errcode∈{0,undefined} 的 ack（超时/errcode 抛错）
  - 收帧分派：req_id∈pending 且 cmd∉回调命令 → resolve 对应 future；cmd∈{aibot_msg_callback,aibot_callback} → onInbound；cmd∈{ping,aibot_event_callback} → 忽略
  - sendRequest：注册 future（键=frame.headers.req_id）→ send → 超时 reject → finally 清理
  - **本任务不含心跳/重连**（Task 4 加），聚焦握手 + 请求/响应相关，用假 socket 测

```ts
import type { Envelope } from "./protocol.js";

const CALLBACK_CMDS = new Set(["aibot_msg_callback", "aibot_callback"]);
const IGNORE_CMDS = new Set(["ping", "aibot_event_callback"]);
const DEFAULT_TIMEOUT = 15_000;
const HANDSHAKE_TIMEOUT = 20_000;

export interface SocketLike {
  send(data: string): void;
  close(): void;
  on(event: "message" | "open" | "close" | "error", cb: (arg?: unknown) => void): void;
}

export type SocketFactory = (url: string) => SocketLike;

export class BotConnection {
  private socket: SocketLike | null = null;
  private readonly pending = new Map<string, { resolve: (e: Envelope) => void; reject: (err: Error) => void }>();

  constructor(
    private readonly opts: {
      url: string;
      botId: string;
      secret: string;
      deviceId: string;
      socketFactory: SocketFactory;
      uuid: () => string;
      onInbound: (env: Envelope) => void;
      onError?: (e: Error) => void;
    },
  ) {}

  connect(): Promise<void> {
    const socket = this.opts.socketFactory(this.opts.url);
    this.socket = socket;
    const subReqId = `subscribe-${this.opts.uuid()}`;

    return new Promise<void>((resolve, reject) => {
      const timer = setTimeout(() => reject(new Error("handshake timeout")), HANDSHAKE_TIMEOUT);
      let handshaken = false;

      socket.on("open", () => {
        socket.send(
          JSON.stringify({
            cmd: "aibot_subscribe",
            headers: { req_id: subReqId },
            body: { bot_id: this.opts.botId, secret: this.opts.secret, device_id: this.opts.deviceId },
          }),
        );
      });

      socket.on("message", (raw) => {
        let env: Envelope;
        try {
          env = JSON.parse(String(raw)) as Envelope;
        } catch {
          return;
        }
        if (!handshaken) {
          if (env.headers?.req_id !== subReqId) return; // 跳过 ping / 无关预认证帧
          const errcode = (env.body as { errcode?: number }).errcode;
          if (errcode !== undefined && errcode !== 0) {
            clearTimeout(timer);
            reject(new Error(`subscribe failed: errcode ${errcode} ${(env.body as { errmsg?: string }).errmsg ?? ""}`));
            return;
          }
          handshaken = true;
          clearTimeout(timer);
          resolve();
          return;
        }
        this.dispatch(env);
      });

      socket.on("error", (e) => {
        const err = e instanceof Error ? e : new Error(String(e));
        if (!handshaken) {
          clearTimeout(timer);
          reject(err);
        } else {
          this.failPending(err);
          this.opts.onError?.(err);
        }
      });

      socket.on("close", () => {
        const err = new Error("socket closed");
        if (!handshaken) {
          clearTimeout(timer);
          reject(err);
        } else {
          this.failPending(err);
          this.opts.onError?.(err);
        }
      });
    });
  }

  private dispatch(env: Envelope): void {
    const reqId = env.headers?.req_id;
    if (reqId && this.pending.has(reqId) && !CALLBACK_CMDS.has(env.cmd)) {
      this.pending.get(reqId)?.resolve(env);
      return;
    }
    if (CALLBACK_CMDS.has(env.cmd)) {
      this.opts.onInbound(env);
      return;
    }
    if (IGNORE_CMDS.has(env.cmd)) return;
    // 其它未知帧：忽略
  }

  private failPending(err: Error): void {
    for (const [, p] of this.pending) p.reject(err);
    this.pending.clear();
  }

  sendFrame(frame: Envelope): void {
    if (!this.socket) throw new Error("not connected");
    this.socket.send(JSON.stringify(frame));
  }

  sendRequest(frame: Envelope, timeoutMs = DEFAULT_TIMEOUT): Promise<Envelope> {
    const reqId = frame.headers.req_id;
    return new Promise<Envelope>((resolve, reject) => {
      const timer = setTimeout(() => {
        this.pending.delete(reqId);
        reject(new Error(`request timeout: ${frame.cmd}`));
      }, timeoutMs);
      this.pending.set(reqId, {
        resolve: (e) => {
          clearTimeout(timer);
          this.pending.delete(reqId);
          resolve(e);
        },
        reject: (err) => {
          clearTimeout(timer);
          this.pending.delete(reqId);
          reject(err);
        },
      });
      this.sendFrame(frame);
    });
  }

  close(): void {
    this.socket?.close();
    this.socket = null;
  }
}
```

```ts
import { randomUUID } from "node:crypto";
import type { ChannelPort, InboundMessage, OutboundMessage } from "@hermes-ts/kernel";
import { inboundMessage } from "@hermes-ts/kernel";
import { BotConnection, type SocketFactory } from "./connection.js";
import { newReqId, parseInbound } from "./protocol.js";
import { ReplyTracker } from "./reply-tracker.js";
import { wsSocketFactory } from "./ws-socket.js";

export type WecomBotConfig = { botId: string; secret: string; url: string };

const BACKOFF = [2000, 5000, 10_000, 30_000, 60_000];
const PING_INTERVAL = 30_000;
const DEDUP_TTL = 300_000;

export class WecomBotChannel implements ChannelPort {
  readonly channel = "wecom-bot";
  private readonly config: WecomBotConfig;
  private readonly socketFactory: SocketFactory;
  private readonly uuid: () => string;
  private readonly deviceId: string;
  private readonly now: () => number;
  private readonly dedupTtlMs: number;
  private readonly replies = new ReplyTracker();
  private readonly seen = new Map<string, number>();
  private connection: BotConnection | null = null;
  private pingTimer: ReturnType<typeof setInterval> | null = null;
  private stopped = false;
  private handler: ((m: InboundMessage) => Promise<void>) | null = null;

  constructor(config: WecomBotConfig, opts: { socketFactory?: SocketFactory; uuid?: () => string; deviceId?: string; dedupTtlMs?: number; now?: () => number } = {}) {
    this.config = config;
    this.socketFactory = opts.socketFactory ?? wsSocketFactory;
    this.uuid = opts.uuid ?? randomUUID;
    this.deviceId = opts.deviceId ?? randomUUID();
    this.now = opts.now ?? Date.now;
    this.dedupTtlMs = opts.dedupTtlMs ?? DEDUP_TTL;
  }

  async start(handler: (m: InboundMessage) => Promise<void>): Promise<void> {
    this.handler = handler;
    await this.openWithRetry(true);
  }

  private async openWithRetry(first: boolean): Promise<void> {
    let attempt = 0;
    for (;;) {
      if (this.stopped) return;
      try {
        await this.openOnce();
        return; // 首次连上即 resolve start()；后续断线由 onError 触发重连循环
      } catch (err) {
        if (this.stopped) return;
        if (first && attempt === 0) throw err; // 首次连接失败让 start() 抛错（配置/凭据错早暴露）
        const delay = BACKOFF[Math.min(attempt, BACKOFF.length - 1)] ?? 60_000;
        await new Promise((r) => setTimeout(r, delay));
        attempt++;
      }
    }
  }

  private async openOnce(): Promise<void> {
    const connection = new BotConnection({
      url: this.config.url,
      botId: this.config.botId,
      secret: this.config.secret,
      deviceId: this.deviceId,
      socketFactory: this.socketFactory,
      uuid: this.uuid,
      onInbound: (env) => void this.onInbound(env),
      onError: () => this.onDisconnect(),
    });
    await connection.connect();
    this.connection = connection;
    this.startPing();
  }

  private onDisconnect(): void {
    this.stopPing();
    this.connection = null;
    if (this.stopped) return;
    void this.openWithRetry(false).catch(() => undefined);
  }

  private async onInbound(env: Parameters<BotConnection["sendRequest"]>[0]): Promise<void> {
    const parsed = parseInbound(env);
    if (!parsed) return;
    const now = this.now();
    for (const [id, ts] of this.seen) if (now - ts > this.dedupTtlMs) this.seen.delete(id);
    if (this.seen.has(parsed.msgId)) return;
    this.seen.set(parsed.msgId, now);
    this.replies.remember({ msgId: parsed.msgId, chatId: parsed.chatId, reqId: parsed.reqId });
    await this.handler?.(
      inboundMessage({ channel: this.channel, conversationId: parsed.chatId, senderId: parsed.fromUser, text: parsed.text, messageId: parsed.msgId }),
    );
  }

  async send(msg: OutboundMessage): Promise<void> {
    if (!this.connection) throw new Error("wecom-bot not connected");
    const frame = this.replies.frameFor(msg.conversationId, msg.text, () => newReqId("aibot_send_msg", this.uuid));
    await this.connection.sendRequest(frame);
  }

  private startPing(): void {
    this.stopPing();
    this.pingTimer = setInterval(() => {
      try {
        this.connection?.sendFrame({ cmd: "ping", headers: { req_id: newReqId("ping", this.uuid) }, body: {} });
      } catch {
        // 忽略——断线由读循环处理
      }
    }, PING_INTERVAL);
  }

  private stopPing(): void {
    if (this.pingTimer) clearInterval(this.pingTimer);
    this.pingTimer = null;
  }

  async stop(): Promise<void> {
    this.stopped = true;
    this.stopPing();
    this.connection?.close();
    this.connection = null;
  }
}
```
